#!/bin/bash

# Check if the service is active
if systemctl is-active apache2  >/dev/null 2>&1;
then
    echo 'apache_service_status{app="test",env="dev",instance="webserver",job="apache_service_status"} 1' > apache_status.txt # Output 1 if service is active
else
    echo 'apache_service_status{app="test",env="dev",instance="webserver",job="apache_service_status"} 0' > apache_status.txt # Output 0 if service is inactive or not found
fi
